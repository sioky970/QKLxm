# 新闻管理功能快速使用指南

## 一、前端集成指南

### 1. 首页新闻展示

```html
<!-- 新闻列表 -->
<div id="news-list"></div>

<script>
// 获取新闻列表
async function loadNewsList() {
  const response = await fetch('/api/news/list?page=1&page_size=5');
  const result = await response.json();
  
  if (result.type === 'ok') {
    const newsHTML = result.data.list.map(news => `
      <div class="news-item" onclick="viewNews(${news.id})">
        <h3>${news.title}</h3>
        <div class="meta">
          <span>作者：${news.author}</span>
          <span>浏览：${news.view_count}</span>
          <span>时间：${formatTime(news.publish_time)}</span>
        </div>
      </div>
    `).join('');
    
    document.getElementById('news-list').innerHTML = newsHTML;
  }
}

// 查看新闻详情
function viewNews(id) {
  window.location.href = `/news-detail.html?id=${id}`;
}

// 格式化时间
function formatTime(timestamp) {
  const date = new Date(timestamp * 1000);
  return date.toLocaleDateString();
}

// 页面加载时获取新闻
loadNewsList();
</script>
```

### 2. 新闻详情页

```html
<!-- news-detail.html -->
<div id="news-detail"></div>

<script>
async function loadNewsDetail() {
  const params = new URLSearchParams(window.location.search);
  const id = params.get('id');
  
  const response = await fetch(`/api/news/detail?id=${id}`);
  const result = await response.json();
  
  if (result.type === 'ok') {
    const news = result.data;
    document.getElementById('news-detail').innerHTML = `
      <h1>${news.title}</h1>
      <div class="meta">
        <span>作者：${news.author}</span>
        <span>浏览：${news.view_count}</span>
        <span>时间：${formatTime(news.publish_time)}</span>
      </div>
      <div class="content">${news.content}</div>
    `;
  } else {
    alert('新闻不存在或已下架');
  }
}

loadNewsDetail();
</script>
```

```html

<div id="category-nav"></div>

// 加载分类列表
async function loadCategories() {

  const result = await response.json();

async function loadNewsByCategory(categoryId) {

  const result = await response.json();
  
  if (result.type === 'ok') {
    // 渲染新闻列表
    renderNewsList(result.data.list);
  }
}

loadCategories();
</script>
```

## 二、后台管理界面集成

### 1. 新闻列表页面

```html
<!-- admin-news-list.html -->
<div class="admin-panel">
  <h2>新闻管理</h2>
  <button onclick="createNews()">创建新闻</button>
  
  <table id="news-table">
    <thead>
      <tr>
        <th>ID</th>
        <th>标题</th>
        <th>状态</th>
        <th>浏览数</th>
        <th>操作</th>
      </tr>
    </thead>
    <tbody id="news-tbody"></tbody>
  </table>
  
  <div id="pagination"></div>
</div>

<script>
const adminToken = localStorage.getItem('admin_token');

// 加载新闻列表
async function loadAdminNewsList(page = 1) {
  const response = await fetch('/api/admin/news/list', {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'Authorization': `Bearer ${adminToken}`
    },
    body: JSON.stringify({
      page: page,
      page_size: 10
    })
  });
  
  const result = await response.json();
  
  if (result.type === 'ok') {
    renderNewsTable(result.data.list);
    renderPagination(result.data.total, result.data.page, result.data.page_size);
  } else if (result.type === 'unauthorized') {
    alert('请先登录');
    window.location.href = '/admin-login.html';
  }
}

// 渲染表格
function renderNewsTable(newsList) {
  const tbody = document.getElementById('news-tbody');
  tbody.innerHTML = newsList.map(news => `
    <tr>
      <td>${news.id}</td>
      <td>${news.title}</td>
      <td>${news.status === 1 ? '已发布' : '草稿'}</td>
      <td>${news.view_count}</td>
      <td>
        <button onclick="editNews(${news.id})">编辑</button>
        ${news.status === 0 ? 
          `<button onclick="publishNews(${news.id})">发布</button>` : 
          `<button onclick="draftNews(${news.id})">撤回</button>`
        }
        <button onclick="deleteNews(${news.id})">删除</button>
      </td>
    </tr>
  `).join('');
}

// 发布新闻
async function publishNews(id) {
  if (!confirm('确定要发布这篇新闻吗？')) return;
  
  const response = await fetch('/api/admin/news/publish', {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'Authorization': `Bearer ${adminToken}`
    },
    body: JSON.stringify({ id })
  });
  
  const result = await response.json();
  
  if (result.type === 'ok') {
    alert('发布成功');
    loadAdminNewsList();
  } else {
    alert('发布失败：' + result.message);
  }
}

// 设为草稿
async function draftNews(id) {
  if (!confirm('确定要撤回为草稿吗？')) return;
  
  const response = await fetch('/api/admin/news/draft', {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'Authorization': `Bearer ${adminToken}`
    },
    body: JSON.stringify({ id })
  });
  
  const result = await response.json();
  
  if (result.type === 'ok') {
    alert('已设为草稿');
    loadAdminNewsList();
  } else {
    alert('操作失败：' + result.message);
  }
}

// 删除新闻
async function deleteNews(id) {
  if (!confirm('确定要删除这篇新闻吗？此操作不可恢复！')) return;
  
  const response = await fetch(`/api/admin/news/${id}`, {
    method: 'DELETE',
    headers: {
      'Authorization': `Bearer ${adminToken}`
    }
  });
  
  const result = await response.json();
  
  if (result.type === 'ok') {
    alert('删除成功');
    loadAdminNewsList();
  } else {
    alert('删除失败：' + result.message);
  }
}

// 页面加载时获取新闻列表
loadAdminNewsList();
</script>
```

### 2. 新闻编辑页面

```html
<!-- admin-news-edit.html -->
<div class="admin-panel">
  <h2 id="page-title">创建新闻</h2>
  
  <form id="news-form">
    <div class="form-group">
      <label>标题：</label>
      <input type="text" id="title" required>
    </div>
    
    <div class="form-group">
      <label>分类：</label>
      <select id="category_id">
        <!-- 动态加载分类选项 -->
      </select>
    </div>
    
    <div class="form-group">
      <label>作者：</label>
      <input type="text" id="author">
    </div>
    
    <div class="form-group">
      <label>内容：</label>
      <!-- 集成富文本编辑器，如TinyMCE或CKEditor -->
      <textarea id="content" rows="20"></textarea>
    </div>
    
    <div class="form-group">
      <button type="button" onclick="saveNews(0)">保存为草稿</button>
      <button type="button" onclick="saveNews(1)">保存并发布</button>
      <button type="button" onclick="history.back()">取消</button>
    </div>
  </form>
</div>

<script>
const adminToken = localStorage.getItem('admin_token');
const params = new URLSearchParams(window.location.search);
const newsId = params.get('id');

// 加载分类列表
async function loadCategories() {
  const response = await fetch('/api/admin/news-category/list', {
    headers: {
      'Authorization': `Bearer ${adminToken}`
    }
  });
  
  const result = await response.json();
  
  if (result.type === 'ok') {
    const select = document.getElementById('category_id');
    select.innerHTML = result.data.map(cat => 
      `<option value="${cat.id}">${cat.name}</option>`
    ).join('');
  }
}

// 如果是编辑模式，加载新闻详情
async function loadNewsDetail() {
  if (!newsId) return;
  
  document.getElementById('page-title').textContent = '编辑新闻';
  
  const response = await fetch(`/api/admin/news/${newsId}`, {
    headers: {
      'Authorization': `Bearer ${adminToken}`
    }
  });
  
  const result = await response.json();
  
  if (result.type === 'ok') {
    const news = result.data;
    document.getElementById('title').value = news.title;
    document.getElementById('category_id').value = news.category_id;
    document.getElementById('author').value = news.author;
    document.getElementById('content').value = news.content;
  }
}

// 保存新闻
async function saveNews(status) {
  const newsData = {
    title: document.getElementById('title').value,
    content: document.getElementById('content').value,
    category_id: parseInt(document.getElementById('category_id').value),
    author: document.getElementById('author').value,
    status: status
  };
  
  if (!newsData.title || !newsData.content) {
    alert('标题和内容不能为空');
    return;
  }
  
  const url = newsId ? `/api/admin/news/${newsId}` : '/api/admin/news/create';
  const method = newsId ? 'PUT' : 'POST';
  
  const response = await fetch(url, {
    method: method,
    headers: {
      'Content-Type': 'application/json',
      'Authorization': `Bearer ${adminToken}`
    },
    body: JSON.stringify(newsData)
  });
  
  const result = await response.json();
  
  if (result.type === 'ok') {
    alert(status === 1 ? '保存并发布成功' : '保存为草稿成功');
    window.location.href = '/admin-news-list.html';
  } else {
    alert('保存失败：' + result.message);
  }
}

// 页面加载
loadCategories();
loadNewsDetail();
</script>
```

```html

<div class="admin-panel">

        <th>状态</th>
        <th>操作</th>
      </tr>
    </thead>
    <tbody id="category-tbody"></tbody>
  </table>
</div>

<!-- 创建/编辑分类模态框 -->
<div id="category-modal" style="display:none;">
  <h3 id="modal-title">创建分类</h3>
  <form id="category-form">
    <input type="hidden" id="edit-id">
    <div class="form-group">
      <label>名称：</label>
      <input type="text" id="cat-name" required>
    </div>
    <div class="form-group">
      <label>描述：</label>
      <input type="text" id="cat-desc">
    </div>
    <div class="form-group">
      <label>状态：</label>
      <select id="cat-status">
        <option value="1">启用</option>
        <option value="0">禁用</option>
      </select>
    </div>
    <button type="button" onclick="saveCategory()">保存</button>
    <button type="button" onclick="hideModal()">取消</button>
  </form>
</div>

<script>
const adminToken = localStorage.getItem('admin_token');

// 加载分类列表
async function loadCategories() {
  const response = await fetch('/api/admin/news-category/list', {
    headers: {
      'Authorization': `Bearer ${adminToken}`
    }
  });
  
  const result = await response.json();
  
  if (result.type === 'ok') {
    const tbody = document.getElementById('category-tbody');
    tbody.innerHTML = result.data.map(cat => `
      <tr>
        <td>${cat.id}</td>
        <td>${cat.name}</td>
        <td>${cat.description}</td>
        <td>${cat.status === 1 ? '启用' : '禁用'}</td>
        <td>
          <button onclick="editCategory(${cat.id}, '${cat.name}', '${cat.description}', ${cat.status})">编辑</button>
          <button onclick="deleteCategory(${cat.id})">删除</button>
        </td>
      </tr>
    `).join('');
  }
}

// 显示创建分类模态框
function showCreateCategoryModal() {
  document.getElementById('modal-title').textContent = '创建分类';
  document.getElementById('edit-id').value = '';
  document.getElementById('cat-name').value = '';
  document.getElementById('cat-desc').value = '';
  document.getElementById('cat-status').value = '1';
  document.getElementById('category-modal').style.display = 'block';
}

// 编辑分类
function editCategory(id, name, desc, status) {
  document.getElementById('modal-title').textContent = '编辑分类';
  document.getElementById('edit-id').value = id;
  document.getElementById('cat-name').value = name;
  document.getElementById('cat-desc').value = desc;
  document.getElementById('cat-status').value = status;
  document.getElementById('category-modal').style.display = 'block';
}

// 保存分类
async function saveCategory() {
  const id = document.getElementById('edit-id').value;
  const categoryData = {
    name: document.getElementById('cat-name').value,
    description: document.getElementById('cat-desc').value,
    status: parseInt(document.getElementById('cat-status').value)
  };
  
  if (!categoryData.name) {
    alert('分类名称不能为空');
    return;
  }
  
  const url = id ? `/api/admin/news-category/${id}` : '/api/admin/news-category/create';
  const method = id ? 'PUT' : 'POST';
  
  const response = await fetch(url, {
    method: method,
    headers: {
      'Content-Type': 'application/json',
      'Authorization': `Bearer ${adminToken}`
    },
    body: JSON.stringify(categoryData)
  });
  
  const result = await response.json();
  
  if (result.type === 'ok') {
    alert('保存成功');
    hideModal();
    loadCategories();
  } else {
    alert('保存失败：' + result.message);
  }
}

// 删除分类
async function deleteCategory(id) {
  if (!confirm('确定要删除这个分类吗？')) return;
  
  const response = await fetch(`/api/admin/news-category/${id}`, {
    method: 'DELETE',
    headers: {
      'Authorization': `Bearer ${adminToken}`
    }
  });
  
  const result = await response.json();
  
  if (result.type === 'ok') {
    alert('删除成功');
    loadCategories();
  } else {
    alert('删除失败：' + result.message);
  }
}

// 隐藏模态框
function hideModal() {
  document.getElementById('category-modal').style.display = 'none';
}

// 页面加载
loadCategories();
</script>
```

## 三、常见问题解决

### 1. 跨域问题

如果前端和后端不在同一域名，需要在后端配置CORS：

后端已配置middleware.Cors()中间件，通常不需要额外配置。

### 2. Token认证

管理员登录后需要保存Token：

```javascript
// 登录成功后
localStorage.setItem('admin_token', token);

// 之后的请求都带上Token
headers: {
  'Authorization': 'Bearer ' + localStorage.getItem('admin_token')
}
```

## 五、注意事项

1. **安全性**: 后台所有API都需要管理员Token认证
2. **XSS防护**: 显示用户输入的内容时要进行HTML转义
3. **图片处理**: 新闻内容中的图片需要单独上传，建议使用OSS或CDN
4. **性能优化**: 对于访问量大的新闻列表，建议添加Redis缓存
5. **SEO优化**: 新闻详情页应该支持服务端渲染（SSR）
